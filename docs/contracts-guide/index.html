核心黑科技一：为什么 CREATE2 能不连区块链、0 成本算出未来合约地址？

早期以太坊部署合约用 CREATE 指令，地址由 部署者地址 + 部署者当前第几笔交易(Nonce) 决定——根本没法提前给 1 万个订单分配地址，因为你不知道哪个订单先付钱部署！
于是以太坊升级引入了 CREATE2：它把交易序号 Nonce 踢掉了，改成由四个完全固定已知的参数做一次哈希运算：

STEP 05

拆开看源码：我们写的 4 个合约到底长什么样？（其实每个核心只有二三十行！）

别被“智能合约工程”吓到，切换下面 4 个标签页看看精简后的核心 Solidity 逻辑——跟写 Go 结构体和接口一样直白：

职责： 全场最简单的合约！只部署 1 份在链上当“母版”，所有订单的 45 字节克隆代理都借用它的代码执行。没有 initialize() 函数，天然免疫代理未初始化被黑客抢注漏洞。 

contract DepositVault {
    // immutable 表示部署母版时直接烧死在字节码里，读它不花 Storage Gas！
    address public immutable SWEEPER;

    constructor(address _sweeper) {
        SWEEPER = _sweeper;
    }

    // 核心归集函数：只有 BatchSweeper 总管能调用！
    function sweep(address token, address coldWallet, uint256 exactAmount) external {
        require(msg.sender == SWEEPER, "Only BatchSweeper allowed");

        // 用内联汇编安全调用 token.transfer(coldWallet, exactAmount)：
        // 1. 兼容以太坊主网 USDT 不返回 bool 的历史奇葩写法；
        // 2. 最多只读取前 32 字节返回数据，防止恶意合约返回几 MB 垃圾数据炸光我们的 Gas (ReturnBomb)。
        _safeTransferBounded(token, coldWallet, exactAmount);
    }
}

职责： ① 提供 predictVaultAddress(salt) 视图函数并与 Go 后端对齐字节码哈希；② 归集时按需在目标地址部署 45 字节的 EIP-1167 最小代理外壳。 

contract VaultFactory {
    address public immutable sweeper;
    address public immutable implementation; // 指向 DepositVault 母版

    constructor(address relayer, address coldWallet, address timelock) {
        // 在一个构造函数里依次创建总管与母版，彻底消除循环地址依赖！
        sweeper = address(new BatchSweeper(address(this), relayer, coldWallet, timelock));
        implementation = address(new DepositVault(sweeper));
    }

    // 0 Gas 纯计算：预测某个订单 salt 对应的充值地址
    function predictVaultAddress(bytes32 salt) public view returns (address) {
        return Clones.predictDeterministicAddress(implementation, salt, address(this));
    }

    // 归集首次触发时，把 45 字节的最小代理部署到预计算好的地址上
    function deployProxy(bytes32 salt) external returns (address) {
        return Clones.cloneDeterministic(implementation, salt);
    }
}

职责： 一笔交易循环处理最多 50 个充值地址。用两步分离 + try/catch 保证哪怕某一个地址被 Tether 拉黑冻结，其余 49 个地址照常归集进冷钱包！而且支持 48 小时时间锁更换热钱包/冷钱包，完全不影响历史订单地址。 

contract BatchSweeper {
    struct SweepItem { bytes32 salt; uint256 exactAmount; }

    function deployAndSweepBatch(SweepItem[] calldata items, address token, address targetCold) external {
        require(msg.sender == relayerHotWallet, "Only Relayer");
        require(targetCold == coldWallet, "Must sweep to ColdWallet");

        for (uint256 i = 0; i < items.length; i++) {
            address vault = factory.predictVaultAddress(items[i].salt);

            // 第一步：如果这个地址还没部署过代理壳，现在原地部署（二次充值的老地址会自动跳过这步！）
            if (vault.code.length == 0) {
                factory.deployProxy(items[i].salt);
            }

            // 第二步：限额 120,000 Gas 尝试抽水，加 try/catch 隔离单地址黑名单报错！
            try IDepositVault(vault).sweep{gas: 120_000}(token, targetCold, items[i].exactAmount) {
                emit VaultSwept(vault, items[i].salt, token, items[i].exactAmount);
            } catch {
                // 单个地址失败不回滚整笔交易，抛出隔离事件通知 Go 后端标为 is_quarantined
                emit VaultQuarantined(vault, items[i].salt, token, items[i].exactAmount);
            }
        }
    }
}

职责： 专门服务于支持 EIP-3009 的原生 USDC（比如 Arbitrum 上的原生 USDC）。用户钱包里没有 ETH 也能付款：用户在网页免费签个名，我们的 Relayer 调这个合约代付几美分 Gas，把用户的 USDC 扣过来直接塞进该订单的 DepositVault，后续跟普通转账走完全同一套扫块入账逻辑！ 

contract EIP3009PayRouter {
    function payOrderWithAuthorization(
        bytes32 salt, address payer, uint256 amount, uint256 validBefore,
        bytes32 authNonce, uint8 v, bytes32 r, bytes32 s
    ) external {
        // 合约自己在链上通过工厂算出收款保险箱地址，哪怕后端被黑也无法把用户的钱拐到黑客地址！
        address vault = vaultFactory.predictVaultAddress(salt);
        require(authNonce == keccak256(abi.encode(block.chainid, salt)), "Invalid nonce");

        // 1. 用 receiveWithAuthorization 把用户授权的 USDC 拉进当前路由合约（防路人抢跑）
        usdc.receiveWithAuthorization(payer, address(this), amount, 0, validBefore, authNonce, v, r, s);

        // 2. 同一笔交易内，立刻全额转入该订单的专属保险箱 vault，触发标准 Transfer 事件供 Go 扫块入账！
        SafeERC20.safeTransfer(IERC20(address(usdc)), vault, amount);
    }
}

结合你最常用的 Arbitrum 与 BSC：真实多链开发必考的 4 个差异坑点

你平时自己用 Arbitrum 和 BSC 最多，这简直是面试时的天然优势！很多没真用过链的候选人以为“都是 EVM 链，配置一模一样”，实际上 Arbitrum 和 BSC 的 USDT/USDC 有几个极容易踩爆的真坑，我们的架构已经全部提前兼容了：

对比维度 Arbitrum (ETH Layer 2, ChainID: 42161) BSC 币安智能链 (L1 PoSA, ChainID: 56)
1. 稳定币小数位精度
(面试头号天坑！)		 USDT 与原生 USDC 都是 6 位小数！
100 USDT = 100,000,000（108）。 		BSC 上的 Binance-Peg USDT 和 USDC 全是 18 位小数！
100 USDT = 100 * 10^18！如果后端写死除以 106，用户充 100U 会被系统算成充了 100 万亿 U！
我们的解法 数据库存原始整数 NUMERIC(78,0) + supported_assets.decimals 按链配置。
2. USDC 免 Gas 签名
(EIP-3009 支持度)		 Circle 官方发行的原生 USDC，完美支持 EIP-3009，收银台自动开启「连接钱包 0 ETH 免手续费一键付」。 BSC 上的 USDC 是币安跨链桥映射的 BEP-20 合约，不支持 Circle 原生 EIP-3009。
我们的解法 supports_eip3009 = false，收银台切到 BSC 时自动隐藏免 Gas 按钮，只展示普通转账。
3. 出块速度与防分叉
(Reorg 确认策略)		 约 0.25 秒出一个软确认块！如果傻傻只数“12 个块”，才过了 3 秒，Sequencer 未提交回 L1 前并不可靠。
我们的解法 区分实时检测（秒级提示用户已看到交易）与最终安全高度（safe 标签入账）。 		Parlia 共识快速终局（Fast Finality），但历史上发生过短分叉重组；通常等待 15 个区块（约 11~45 秒）或 finalized 标签 即可安全入账。
4. 合约层兼容性 100% 共用同一套 Solidity 合约！ VaultFactory、DepositVault、BatchSweeper 在 Arbitrum 和 BSC 上一行代码都不用改，分别部署一份即可同时运转。
